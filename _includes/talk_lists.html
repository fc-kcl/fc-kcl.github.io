{%- comment -%} Splits _data/talks.yml into two lists, both available to the including page: upcoming talks dated today or later, soonest first (undated talks last) past talks dated before today, most recent first {%- endcomment -%} {%- comment -%} `site.data.talks` is false when _data/talks.yml is empty or fully commented out, and Liquid's sort filter cannot handle that, so default to an empty list. {%- endcomment -%} {%- assign today = 'now' | date: '%Y-%m-%d' -%} {%- assign empty_list = '' | split: ',' -%} {%- assign all_talks = site.data.talks | default: empty_list -%} {%- assign by_date = all_talks | sort: 'date' -%} {%- assign dated_upcoming = by_date | where_exp: "t", "t.date >= today" -%} {%- assign undated = by_date | where_exp: "t", "t.date == nil" -%} {%- assign upcoming = dated_upcoming | concat: undated -%} {%- assign past = by_date | where_exp: "t", "t.date < today" | reverse -%}
